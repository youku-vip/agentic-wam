import sqlite3
from collections import defaultdict

db = sqlite3.connect("/tmp/aw0_cuda_trace.sqlite")
query = """
SELECT
    k.deviceId,
    k.start,
    k.end,
    s.value
FROM CUPTI_ACTIVITY_KIND_KERNEL AS k
JOIN StringIds AS s ON k.demangledName = s.id
WHERE LOWER(s.value) LIKE '%nccl%'
ORDER BY k.start
"""
rows = db.execute(query).fetchall()

t0 = db.execute(
    "SELECT MIN(start) FROM CUPTI_ACTIVITY_KIND_KERNEL"
).fetchone()[0]

print("=== NCCL kernels by GPU ===")
groups = defaultdict(list)
for gpu, start, end, name in rows:
    groups[gpu].append((start, end))

for gpu, items in sorted(groups.items()):
    durations = [(e - s) / 1e6 for s, e in items]
    print(
        f"GPU {gpu}: count={len(items)}, "
        f"total={sum(durations):.1f} ms, "
        f"avg={sum(durations)/len(durations):.1f} ms, "
        f"min={min(durations):.1f} ms, "
        f"max={max(durations):.1f} ms"
    )

print("\n=== Individual NCCL kernel timeline ===")
print("GPU  Start(s)  Duration(ms)")
for gpu, start, end, name in rows:
    print(
        f"{gpu:3d}  "
        f"{(start-t0)/1e9:8.3f}  "
        f"{(end-start)/1e6:12.1f}"
    )

db.close()

import os
import json
import torch

cache = torch.load(
    os.environ["CACHE"],
    map_location="cpu",
    weights_only=False,
)
order = json.load(open(os.environ["ORDER"]))

codes = cache["codes"]
ids = cache["code_id_by_idx"].reshape(-1).tolist()

print("Episodes:", len(order))
print("Cache frames:", len(ids))
print("Unique codes:", len(codes))

expected_frames = sum(int(x["length"]) for x in order)
assert len(ids) == expected_frames, (
    f"Frame mismatch: cache={len(ids)}, "
    f"order={expected_frames}"
)

assert min(ids) >= 0
assert max(ids) < len(codes)

offset = 0
stats = []
examples = []

for ei, ep in enumerate(order):
    start = int(ep["offset"])
    length = int(ep["length"])

    assert start == offset, (
        f"Episode {ei}: offset={start}, expected={offset}"
    )
    offset += length

    seq = ids[start:start + length]
    transitions = [
        t for t in range(1, length)
        if seq[t] != seq[t - 1]
    ]
    stats.append(len(transitions))

    if transitions and len(examples) < 12:
        t = transitions[0]
        examples.append((ei, t, seq[t - 1], seq[t]))

print("Total Code transitions:", sum(stats))
print("Episodes with no Code transition:",
      sum(n == 0 for n in stats))
print("Max transitions in one episode:", max(stats))
print("Mean transitions per episode:",
      round(sum(stats) / len(stats), 2))

print("\nExample subtask transitions:")
for ei, t, old_id, new_id in examples:
    print(f"\nEpisode {ei}, local frame {t}")
    print("BEFORE:", repr(codes[old_id][:250]))
    print("AFTER: ", repr(codes[new_id][:250]))

print("\nINDEX_AND_BOUNDARY_SCAN_OK")
PY
